import { useState, useEffect } from "react";
import { CPLogin } from "./components/cardpath/CPLogin";
import { Quiz } from "./components/cardpath/Quiz";
import { CardPathApp } from "./components/cardpath/CardPathApp";
import { loadProfile, UserProfile } from "./components/cardpath/data";

type AppState = "login" | "quiz" | "app";

export default function App() {
  const [state, setState] = useState<AppState>("login");
  const [profile, setProfile] = useState<UserProfile | null>(null);
  const [hasExisting, setHasExisting] = useState(false);

  useEffect(() => {
    const saved = loadProfile();
    if (saved?.onboardingComplete) {
      setProfile(saved);
      setHasExisting(true);
    } else if (saved) {
      setHasExisting(true);
    }
  }, []);

  const handleLogin = () => {
    if (profile) {
      setState("app");
    } else {
      setState("quiz");
    }
  };

  const handleSignUp = () => {
    setState("quiz");
  };

  const handleRestore = () => {
    const saved = loadProfile();
    if (saved?.onboardingComplete) {
      setProfile(saved);
      setState("app");
    }
  };

  const handleQuizComplete = (completed: UserProfile) => {
    setProfile(completed);
    setState("app");
  };

  const handleProfileUpdate = (updated: UserProfile) => {
    setProfile(updated);
  };

  const handleRetakeQuiz = () => {
    setState("quiz");
  };

  if (state === "login") {
    return (
      <CPLogin
        onLogin={handleLogin}
        onSignUp={handleSignUp}
        hasExisting={hasExisting}
        onRestore={handleRestore}
      />
    );
  }

  if (state === "quiz") {
    return <Quiz onComplete={handleQuizComplete} />;
  }

  return (
    <CardPathApp
      profile={profile!}
      onProfileUpdate={handleProfileUpdate}
      onRetakeQuiz={handleRetakeQuiz}
    />
  );
}

import { useState } from "react";
import { CPSidebar } from "./CPSidebar";
import { CPDashboard } from "./CPDashboard";
import { CardExplorer } from "./CardExplorer";
import { PathGraph } from "./PathGraph";
import { SpendingAnalysis } from "./SpendingAnalysis";
import { SpendingSimulator } from "./SpendingSimulator";
import { AIAdvisor } from "./AIAdvisor";
import { CPSettings } from "./CPSettings";
import { UserProfile } from "./data";

interface CardPathAppProps {
  profile: UserProfile;
  onProfileUpdate: (p: UserProfile) => void;
  onRetakeQuiz: () => void;
}

export function CardPathApp({ profile, onProfileUpdate, onRetakeQuiz }: CardPathAppProps) {
  const [currentPage, setCurrentPage] = useState("dashboard");

  const renderPage = () => {
    switch (currentPage) {
      case "dashboard": return <CPDashboard profile={profile} onNavigate={setCurrentPage} />;
      case "explorer": return <CardExplorer profile={profile} />;
      case "pathgraph": return <PathGraph profile={profile} />;
      case "spending": return <SpendingAnalysis profile={profile} onNavigate={setCurrentPage} />;
      case "simulator": return <SpendingSimulator profile={profile} />;
      case "advisor": return <AIAdvisor profile={profile} />;
      case "settings": return <CPSettings profile={profile} onProfileUpdate={onProfileUpdate} onRetakeQuiz={onRetakeQuiz} />;
      default: return <CPDashboard profile={profile} onNavigate={setCurrentPage} />;
    }
  };

  return (
    <div className="flex h-screen bg-background">
      <div className="flex-shrink-0">
        <CPSidebar currentPage={currentPage} onPageChange={setCurrentPage} />
      </div>
      <main className="flex-1 overflow-auto">
        {renderPage()}
      </main>
    </div>
  );
}
